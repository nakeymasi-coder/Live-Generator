function esc(s){return s.replace(/[&<>"']/g,m=>({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"}[m]))}


function componentGroups(prompt){
  const clauses = prompt
    .replace(/\n+/g,' ')
    .split(/(?<=[.!?])\s+|;\s+|,\s+(?=(?:and\s+)?(?:dress|style|give|add|place|pose|use|background|lighting|hair|makeup|wear|show)\b)/i)
    .map(x=>x.trim())
    .filter(x=>x.length>18);

  const groups={};
  const add=(g,t)=>{ if(!groups[g]) groups[g]=[]; if(!groups[g].includes(t)) groups[g].push(t); };

  clauses.forEach(c=>{
    const l=c.toLowerCase();
    let g="Details";
    if(/hair|braid|curl|ponytail|bun|bang|baby hair|waves|locs|pigtail/.test(l)) g="Hair";
    else if(/eye|lash|lip|makeup|brow|cheek|skin|nose|face|glossy|complexion/.test(l)) g="Face + Makeup";
    else if(/dress|wear|top|shirt|jersey|skirt|jeans|jogger|cardigan|shorts|outfit|sweatshirt|dress her/.test(l)) g="Outfit";
    else if(/earring|necklace|watch|bracelet|ring|jewelry|headphone|glasses|ornament|butterfl|flower/.test(l)) g="Accessories";
    else if(/pose|seated|sitting|standing|hand|reclining|walking|cross-legged|shoulder/.test(l)) g="Pose";
    else if(/background|garden|studio|room|interior|sofa|bedroom|café|rooftop|forest|setting/.test(l)) g="Background";
    else if(/light|glow|rim|shadow|bokeh|depth of field|cinematic|illumination|sunset|daylight/.test(l)) g="Lighting";
    else if(/render|illustration|editorial|realism|3d|digital art|texture|finish|campaign|quality/.test(l)) g="Rendering";
    add(g,c.replace(/\.$/,''));
  });

  return groups;
}

function renderPresetComponents(refId,presetIndex,prompt){
  const groups=componentGroups(prompt);
  return Object.entries(groups).map(([group,items])=>`
    <div class="component-title">${esc(group)}</div>
    ${items.map((text,idx)=>`
      <label class="component-choice">
        <input type="checkbox" class="componentcheck"
          data-ref="${refId}" data-index="${presetIndex}" data-group="${esc(group)}"
          value="${esc(text)}">
        <span>${esc(text)}</span>
      </label>`).join('')}
  `).join('');
}

function buildPrompt(remix=false){
  const refs=[...document.querySelectorAll('.refcheck:checked')].map(x=>Number(x.value));
  const presetNodes=[...document.querySelectorAll('.presetcheck:checked')];
  let presetPrompts=presetNodes.map(n=>REFS.find(r=>r.id===Number(n.dataset.ref)).presets[Number(n.dataset.index)].prompt);
  if(remix && presetPrompts.length>1) presetPrompts=presetPrompts.sort(()=>Math.random()-.5);

  const styles=selectedValues('stylecheck');
  const components=selectedValues('componentcheck');
  const types=selectedValues('typecheck');
  const products=selectedValues('productcheck');
  const exact=document.getElementById('textExact').value.trim();
  const notes=document.getElementById('productNotes').value.trim();

  let pieces=[];
  if(refs.length) pieces.push(`Use the selected visual references as inspiration: ${refs.map(i=>'Image '+i).join(', ')}. Blend their visual qualities into one cohesive original design rather than copying any single image exactly.`);
  if(presetPrompts.length) pieces.push(`Blend these selected preset directions:\n\n${presetPrompts.map((p,i)=>`${i+1}. ${p}`).join('\n\n')}`);
  if(components.length) pieces.push(`Selected mix-and-match preset pieces: ${components.join('; ')}.`);
  if(styles.length) pieces.push(`Additional selected style ingredients: ${styles.join('; ')}.`);
  if(exact || types.length){
    let t=`Typography layer: `;
    if(exact) t+=`include the exact wording “${exact}”. `;
    if(types.length) t+=`Mix these typography directions where they work together: ${types.join('; ')}. `;
    t+=`Keep all visible wording correctly spelled, fully visible, unobstructed, and easy to read.`;
    pieces.push(t);
  }
  if(products.length || notes){
    let p=`Product/design use: `;
    if(products.length) p+=products.join('; ')+'. ';
    if(notes) p+=notes+'. ';
    p+=`Keep the composition practical for the selected product format.`;
    pieces.push(p);
  }
  pieces.push(`Final quality rules: adult subjects only; polished luxury finish; no crowns or tiaras unless explicitly requested; no fake HUDs, dashboards, analytics panels, scan lines, random data graphics, or clutter; preserve requested body type and proportions; keep the design clean, intentional, and professionally branded.`);
  const result=pieces.join('\n\n');
  document.getElementById('promptOutput').value=result;
  document.getElementById('count').textContent=`${result.length.toLocaleString()} characters`;
  updatePills();
  return result;
}

function updatePills(){
  const pills=[];
  document.querySelectorAll('.refcheck:checked').forEach(x=>pills.push('Image '+x.value));
  document.querySelectorAll('.presetcheck:checked').forEach(n=>{
    const r=REFS.find(r=>r.id===Number(n.dataset.ref));
    pills.push(r.presets[Number(n.dataset.index)].name);
  });
  const componentCount=document.querySelectorAll('.componentcheck:checked').length;
  if(componentCount) pills.push(componentCount+' mixed preset piece'+(componentCount===1?'':'s'));
  document.getElementById('selectedPills').innerHTML=pills.map(p=>`<span class="pill">${esc(p)}</span>`).join('');
}
